with first as ( -- the first login date for every player
    select player_id, min(event_date) as m
    from activity
    group by player_id
)
select round(
    (count(a.player_id)) / (select count(*) from first) -- total players unique
    ,2
) as fraction
from activity a
join first f
on a.player_id = f.player_id
where a.event_date = date_add(f.m , interval 1 day)
